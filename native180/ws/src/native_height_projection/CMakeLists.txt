cmake_minimum_required(VERSION 3.8)
project(native_height_projection)
find_package(ament_cmake REQUIRED)
add_library(free_evidence SHARED src/free_evidence.cpp)
target_compile_features(free_evidence PRIVATE cxx_std_14)
target_compile_options(free_evidence PRIVATE -O3 -Wall -Wextra)
install(TARGETS free_evidence LIBRARY DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/native_height_projector.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/projection_core.py scripts/free_evidence.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
ament_package()
